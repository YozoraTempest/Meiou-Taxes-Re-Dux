Redux_SubjectSelection_select = {
    set_province_flag = Pin_Show
    set_province_flag = UI_Select
    set_global_flag = UI_Select
}

Redux_SubjectSelection_deselect = {
    set_province_flag = Pin_Hide
    clr_province_flag = UI_Select
}

Redux_SelectCountryRealm = {
    every_owned_province = {
        limit = {
            $target$ = yes
            Redux_SelectionChange_$action$ = yes
            OR = {
                event_target:Redux_SelectionActor = {
                    is_key_equal = { lhs = UI_SelectScope value = 5 }
                }
                Redux_SelectionState_$action$ = yes
            }
        }
        Redux_SubjectSelection_$action$ = yes
    }
}

Redux_SelectRealm = {
    event_target:Redux_SelectionActor = {
        Redux_SelectCountryRealm = { action = $action$ target = $target$ }
        every_subject_country = {
            limit = {
                is_subject_other_than_tributary_trigger = yes
                is_subject_of = event_target:Redux_SelectionActor
            }
            Redux_SelectCountryRealm = { action = $action$ target = $target$ }
        }
    }
}

# The actor is the player, even when the seed province belongs to a subject.
Redux_ApplySelectionScope = {
    if = {
        limit = { event_target:Redux_SelectionActor = { is_key_equal = { lhs = UI_SelectScope value = 1 } } }
        every_neighbor_province = {
            limit = {
                $target$ = yes
                Redux_SelectionChange_$action$ = yes
            }
            Redux_SubjectSelection_$action$ = yes
        }
    }
    else_if = {
        limit = { event_target:Redux_SelectionActor = { is_key_equal = { lhs = UI_SelectScope value = 2 } } }
        area = {
            limit = {
                $target$ = yes
                Redux_SelectionChange_$action$ = yes
            }
            Redux_SubjectSelection_$action$ = yes
        }
    }
    else_if = {
        limit = { event_target:Redux_SelectionActor = { is_key_equal = { lhs = UI_SelectScope value = 3 } } }
        region = {
            limit = {
                $target$ = yes
                Redux_SelectionChange_$action$ = yes
            }
            Redux_SubjectSelection_$action$ = yes
        }
    }
    else_if = {
        limit = {
            event_target:Redux_SelectionActor = {
                OR = {
                    is_key_equal = { lhs = UI_SelectScope value = 4 }
                    is_key_equal = { lhs = UI_SelectScope value = 5 }
                }
            }
        }
        Redux_SelectRealm = { action = $action$ target = $target$ }
    }
    else_if = {
        limit = { event_target:Redux_SelectionActor = { is_key_equal = { lhs = UI_SelectScope value = 6 } } }
        every_trade_node_member_province = {
            limit = {
                $target$ = yes
                Redux_SelectionChange_$action$ = yes
            }
            Redux_SubjectSelection_$action$ = yes
        }
    }
}

# Native buttons already handle owned provinces and schedule the pin event.
Redux_ExpandSubjectSelection = {
    if = {
        limit = {
            owner = {
                ai = no
                has_country_flag = Redux_IncludeSubjects
                NOT = { has_country_flag = UI_Freeze }
            }
        }
        owner = { save_event_target_as = Redux_SelectionActor }
        Redux_ApplySelectionScope = {
            action = $action$
            target = Redux_SubjectSelectionTarget
        }
    }
}

Redux_ToggleSubjectSelection = {
    if = {
        limit = { Redux_SubjectSelectionButtonAllowed = yes }
        FROM = { save_event_target_as = Redux_SelectionActor }
        if = {
            limit = {
                OR = {
                    has_province_flag = Pin_Show
                    has_province_flag = UI_Select
                }
            }
            Redux_ApplySelectionScope = { action = deselect target = Redux_RealmSelectionTarget }
            Redux_SubjectSelection_deselect = yes
            hidden_effect = { province_event = { id = SYS_Pin.005 } }
        }
        else = {
            Redux_ApplySelectionScope = { action = select target = Redux_RealmSelectionTarget }
            Redux_SubjectSelection_select = yes
            hidden_effect = { province_event = { id = SYS_Pin.006 } }
        }
    }
}

Redux_ExpandSubjectToggle = {
    if = {
        limit = {
            OR = {
                has_province_flag = Pin_Show
                has_province_flag = UI_Select
            }
        }
        Redux_ExpandSubjectSelection = { action = deselect }
    }
    else = {
        Redux_ExpandSubjectSelection = { action = select }
    }
}
